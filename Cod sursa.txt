// =====================================================
// INITIALIZARE CAMERA
// =====================================================

bool initCamera() {

  camera_config_t config;

  config.ledc_channel = LEDC_CHANNEL_0;
  config.ledc_timer = LEDC_TIMER_0;

  config.pin_d0 = Y2_GPIO_NUM;
  config.pin_d1 = Y3_GPIO_NUM;
  config.pin_d2 = Y4_GPIO_NUM;
  config.pin_d3 = Y5_GPIO_NUM;
  config.pin_d4 = Y6_GPIO_NUM;
  config.pin_d5 = Y7_GPIO_NUM;
  config.pin_d6 = Y8_GPIO_NUM;
  config.pin_d7 = Y9_GPIO_NUM;

  config.pin_xclk = XCLK_GPIO_NUM;
  config.pin_pclk = PCLK_GPIO_NUM;
  config.pin_vsync = VSYNC_GPIO_NUM;
  config.pin_href = HREF_GPIO_NUM;

  config.pin_sccb_sda = SIOD_GPIO_NUM;
  config.pin_sccb_scl = SIOC_GPIO_NUM;

  config.pin_pwdn = PWDN_GPIO_NUM;
  config.pin_reset = RESET_GPIO_NUM;

  config.xclk_freq_hz = 20000000;

  config.pixel_format = PIXFORMAT_JPEG;


  // ===================================================
  // SETARI IMAGINE
  // ===================================================

  if (psramFound()) {

    config.frame_size = FRAMESIZE_VGA;
    config.jpeg_quality = 10;
    config.fb_count = 2;

    Serial.println("PSRAM detectat - VGA");

  } else {

    config.frame_size = FRAMESIZE_QVGA;
    config.jpeg_quality = 12;
    config.fb_count = 1;

    Serial.println("Fara PSRAM - QVGA");
  }


  // ===================================================
  // INITIALIZARE
  // ===================================================

  esp_err_t err = esp_camera_init(&config);

  if (err != ESP_OK) {

    Serial.printf(
      "Eroare initializare camera: 0x%x\n",
      err
    );

    return false;
  }


  Serial.println(
    "Camera initializata cu succes."
  );

  return true;
}

void connectWiFi() {

  Serial.println();
  Serial.println("Conectare la Wi-Fi...");

  WiFi.mode(WIFI_STA);

  // CORECTAT:
  // folosim ssid si password, exact cum sunt definite sus
  WiFi.begin(
    ssid,
    password
  );


  while (WiFi.status() != WL_CONNECTED) {

    delay(500);

    Serial.print(".");
  }


  Serial.println();

  Serial.println(
    "Wi-Fi conectat!"
  );

  Serial.print(
    "IP Camera X: "
  );

  Serial.println(
    WiFi.localIP()
  );
}

// =====================================================
// LIVE STREAM
// =====================================================

void handleStream() {

  WiFiClient client = server.client();


  client.println(
    "HTTP/1.1 200 OK"
  );

  client.println(
    "Content-Type: multipart/x-mixed-replace; boundary=frame"
  );

  client.println(
    "Cache-Control: no-cache"
  );

  client.println(
    "Access-Control-Allow-Origin: *"
  );

  client.println(
    "Connection: close"
  );

  client.println();


  // ===================================================
  // TRIMITERE CADRE
  // ===================================================

  while (client.connected()) {

    camera_fb_t* fb =
      esp_camera_fb_get();


    if (!fb) {

      Serial.println(
        "Eroare captura stream!"
      );

      break;
    }


    client.println(
      "--frame"
    );

    client.println(
      "Content-Type: image/jpeg"
    );

    client.println(
      "Content-Length: " +
      String(fb->len)
    );

    client.println();


    client.write(
      fb->buf,
      fb->len
    );


    client.println();


    esp_camera_fb_return(fb);


    delay(40);
  }


  client.stop();
}


// =====================================================
// TELEGRAM - TRIMITERE FOTO
// =====================================================

bool sendPhotoTelegram(
  camera_fb_t* fb
) {

  WiFiClientSecure client;

  // Pentru prototip
  client.setInsecure();


  const char* host =
    "api.telegram.org";


  Serial.println(
    "Conectare la Telegram..."
  );


  if (!client.connect(
        host,
        443
      )) {

    Serial.println(
      "Eroare conectare Telegram!"
    );

    return false;
  }


  // ===================================================
  // MULTIPART BOUNDARY
  // ===================================================

  String boundary =
    "----ESP32CAMBoundary";


  // ===================================================
  // HEADER MULTIPART
  // ===================================================

  String head =
    "--" + boundary + "\r\n"

    "Content-Disposition: form-data; "
    "name=\"chat_id\"\r\n\r\n"

    + String(CHAT_ID) +

    "\r\n"

    "--" + boundary + "\r\n"

    "Content-Disposition: form-data; "
    "name=\"caption\"\r\n\r\n"

    "ALERTA DE MISCARE - Camera X\r\n"

    "--" + boundary + "\r\n"

    "Content-Disposition: form-data; "
    "name=\"photo\"; "
    "filename=\"cameraX.jpg\"\r\n"

    "Content-Type: image/jpeg\r\n\r\n";


  // ===================================================
  // FINAL MULTIPART
  // ===================================================

  String tail =
    "\r\n--" +
    boundary +
    "--\r\n";


  // ===================================================
  // URL TELEGRAM
  // ===================================================

  String url =
    "/bot" +
    String(BOT_TOKEN) +
    "/sendPhoto";


  // ===================================================
  // LUNGIME CONTINUT
  // ===================================================

  uint32_t contentLength =
    head.length() +
    fb->len +
    tail.length();


  // ===================================================
  // HTTP POST
  // ===================================================

  client.println(
    "POST " +
    url +
    " HTTP/1.1"
  );


  client.println(
    "Host: " +
    String(host)
  );


  client.println(
    "Content-Type: multipart/form-data; boundary=" +
    boundary
  );


  client.println(
    "Content-Length: " +
    String(contentLength)
  );


  client.println(
    "Connection: close"
  );


  client.println();


  // ===================================================
  // HEADER
  // ===================================================

  client.print(head);


  // ===================================================
  // FOTOGRAFIA
  // ===================================================

  client.write(
    fb->buf,
    fb->len
  );


  // ===================================================
  // FINAL
  // ===================================================

  client.print(tail);


  // ===================================================
  // CITIRE RASPUNS TELEGRAM
  // ===================================================

  unsigned long timeout =
    millis();

  bool success = false;


  while (
    client.connected() &&
    millis() - timeout < 10000
  ) {

    while (client.available()) {

      String line =
        client.readStringUntil('\n');

      Serial.println(line);


      if (
        line.startsWith(
          "HTTP/1.1 200"
        )
      ) {

        success = true;
      }


      timeout = millis();
    }
  }


  client.stop();


  return success;
}


// =====================================================
// CAPTURA SI TRIMITERE TELEGRAM
// =====================================================

void captureAndSend() {

  Serial.println();
  Serial.println(
    "Capturare fotografie..."
  );


  camera_fb_t* fb =
    esp_camera_fb_get();


  if (!fb) {

    Serial.println(
      "Eroare la capturarea fotografiei!"
    );

    return;
  }


  Serial.print(
    "Dimensiune fotografie: "
  );

  Serial.print(
    fb->len
  );

  Serial.println(
    " bytes"
  );


  bool result =
    sendPhotoTelegram(fb);


  if (result) {

    Serial.println(
      "Fotografia a fost trimisa pe Telegram!"
    );

  } else {

    Serial.println(
      "Fotografia NU a fost trimisa!"
    );
  }


  esp_camera_fb_return(fb);
}

void setup() {

  Serial.begin(115200);

  delay(1000);


  Serial.println();

  Serial.println(
    "================================"
  );

  Serial.println(
    " SMART SECURITY CAMERA"
  );

  Serial.println(
    " CAMERA X"
  );

  Serial.println(
    "================================"
  );


  // ===================================================
  // PIR
  // ===================================================

  pinMode(
    PIR_PIN,
    INPUT
  );


  // ===================================================
  // CAMERA
  // ===================================================

  if (!initCamera()) {

    Serial.println(
      "Camera nu a putut fi initializata!"
    );


    while (true) {

      delay(1000);
    }
  }


  // ===================================================
  // WIFI
  // ===================================================

  connectWiFi();


  // ===================================================
  // WEB SERVER
  // ===================================================

  server.on(
    "/",
    HTTP_GET,
    handleRoot
  );


  server.on(
    "/stream",
    HTTP_GET,
    handleStream
  );


  server.begin();


  Serial.println(
    "Server web pornit!"
  );


  Serial.print(
    "Deschide in browser: http://"
  );

  Serial.println(
    WiFi.localIP()
  );


  // ===================================================
  // STABILIZARE PIR
  // ===================================================

  Serial.println(
    "Asteptam stabilizarea PIR..."
  );


  delay(5000);


  Serial.println(
    "SISTEM PREGATIT!"
  );
}


// =====================================================
// LOOP
// =====================================================

void loop() {

  // ===================================================
  // SERVER WEB
  // ===================================================

  server.handleClient();


  // ===================================================
  // PIR
  // ===================================================

  int pirState =
    digitalRead(PIR_PIN);


  if (pirState == HIGH) {

    unsigned long currentTime =
      millis();


    // =================================================
    // DETECTARE NOUA MISCARE
    // =================================================

    if (
      !motionActive &&
      currentTime - lastMotionTime >
      MOTION_COOLDOWN
    ) {

      motionActive = true;


      lastMotionTime =
        currentTime;


      Serial.println();

      Serial.println(
        "================================"
      );

      Serial.println(
        "🚨 MISCARE DETECTATA!"
      );

      Serial.println(
        "Camera X"
      );

      Serial.println(
        "================================"
      );


      captureAndSend();
    }

  } else {

    motionActive = false;
  }


  delay(50);
}
